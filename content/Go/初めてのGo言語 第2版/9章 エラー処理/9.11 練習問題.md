type Employee struct {
	ID        string `json:"id"`
	FirstName string `json:"first_name"`
	LastName  string `json:"last_name"`
	Title     string `json:"title"`
}

var (
	validID = regexp.MustCompile(`\w{4}-\d{3}`)
)

func ValidateEmployee(e Employee) error {
	var errs []error

	if len(e.ID) == 0 {
		errs = append(errs, errors.New("IDがありません"))
	}
	if !validID.MatchString(e.ID) {
		errs = append(errs, errors.New("無効なIDです"))
	}
	if len(e.FirstName) == 0 {
		errs = append(errs, errors.New("名前がありません"))
	}
	if len(e.LastName) == 0 {
		errs = append(errs, errors.New("名字がありません"))
	}
	if len(e.Title) == 0 {
		errs = append(errs, errors.New("タイトルがありません"))
	}
	// エラーの数に応じて、nilを返す場合、ひとつのエラーを返す場合、errors.Joinを使って、ひとつのエラーにすべてのエラーをまとめる場合、に分ける
 	// これはGoのできるだけシンプルにする哲学に従うものである
 	switch len(errs) {
 	case 0:
 		return nil
 	case 1:
 		return errs[0]
 	default:
 		return errors.Join(errs...)
}
